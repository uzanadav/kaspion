### 1.1 `kaspion/db.py` — connection + DDL (single source of truth for raw/state schemas)

```python
from __future__ import annotations

import duckdb
from pathlib import Path

DB_PATH = Path(__file__).resolve().parent.parent / "data" / "finance.duckdb"

DDL = """
CREATE SCHEMA IF NOT EXISTS raw;
CREATE SCHEMA IF NOT EXISTS state;

CREATE TABLE IF NOT EXISTS raw.transactions (
    transaction_id  TEXT PRIMARY KEY,
    account_id      TEXT NOT NULL,
    account_type    TEXT NOT NULL CHECK (account_type IN ('bank', 'credit_card')),
    posted_date     DATE NOT NULL,
    amount          DECIMAL(18, 2) NOT NULL,   -- negative = outflow, positive = inflow
    currency        TEXT NOT NULL DEFAULT 'ILS',
    raw_description TEXT NOT NULL,
    source          TEXT NOT NULL,
    ingested_at     TIMESTAMP NOT NULL DEFAULT current_timestamp
);

CREATE TABLE IF NOT EXISTS state.ai_proposals (
    merchant_key         TEXT PRIMARY KEY,
    proposed_category_id TEXT NOT NULL,
    provider             TEXT NOT NULL,
    model                TEXT NOT NULL,
    confidence           DOUBLE,
    created_at           TIMESTAMP NOT NULL DEFAULT current_timestamp
);

CREATE TABLE IF NOT EXISTS state.merchant_overrides (
    merchant_key TEXT PRIMARY KEY,
    category_id  TEXT NOT NULL,
    updated_at   TIMESTAMP NOT NULL DEFAULT current_timestamp
);

CREATE TABLE IF NOT EXISTS state.budgets (
    category_id        TEXT PRIMARY KEY,
    monthly_amount_ils DECIMAL(18, 2) NOT NULL,
    effective_from     DATE NOT NULL DEFAULT current_date
);
"""


def connect(read_only: bool = False) -> duckdb.DuckDBPyConnection:
    DB_PATH.parent.mkdir(exist_ok=True)
    con = duckdb.connect(str(DB_PATH), read_only=read_only)
    if not read_only:
        con.execute(DDL)
    return con
```

Note: AI proposals are keyed on `merchant_key`, not `transaction_id` — one merchant gets one category, which is what merchant memory means and cuts AI calls to unique-new-merchants only.

### 1.3 `kaspion/ingest/generate_seed.py` — synthetic data generator

```python
"""Generate realistic synthetic Israeli transactions -> dbt/seeds/sample_transactions.csv.

Deliberately includes: Hebrew+English merchants, salary inflows, inter-account
transfer pairs, and monthly credit-card debits that equal each card's statement sum.
"""
from __future__ import annotations

import csv
import hashlib
import random
from collections import defaultdict
from datetime import date, timedelta
from pathlib import Path

OUT = Path(__file__).resolve().parents[2] / "dbt" / "seeds" / "sample_transactions.csv"

ACCOUNTS = {
    "leumi-main": "bank",
    "leumi-savings": "bank",
    "max-1234": "credit_card",
    "isracard-5678": "credit_card",
}
CARD_TO_BANK = {"max-1234": "leumi-main", "isracard-5678": "leumi-main"}
CARD_DEBIT_DESC = {"max-1234": "חיוב מקס איט פיננסים", "isracard-5678": "חיוב ישראכרט"}

# merchant -> (category hint for ground truth, typical amount range)
MERCHANTS = {
    "רמי לוי 342": ("groceries", (150, 700)),
    "שופרסל דיל": ("groceries", (80, 450)),
    "יוחננוף": ("groceries", (100, 500)),
    "וולט": ("restaurants", (60, 180)),
    "קפה גרג": ("restaurants", (30, 90)),
    "מסעדת הבשריה": ("restaurants", (150, 450)),
    "פז יעד לוד": ("transport", (150, 350)),
    "רב קו אונליין": ("transport", (30, 150)),
    "GETT": ("transport", (25, 90)),
    "חברת החשמל": ("housing", (250, 900)),
    "מי אביבים": ("housing", (80, 250)),
    "ארנונה תל אביב": ("housing", (400, 700)),
    "סופר פארם": ("health", (40, 250)),
    "מכבי שירותי בריאות": ("health", (50, 150)),
    "גן ילדים פרטי": ("kids", (2500, 3500)),
    "ZARA TLV": ("clothing", (100, 500)),
    "NETFLIX.COM": ("subscriptions", (55, 55)),
    "SPOTIFY": ("subscriptions", (25, 25)),
    "iCloud": ("subscriptions", (12, 12)),
    "KSP מחשבים": ("electronics", (100, 1500)),
    "סינמה סיטי": ("entertainment", (80, 200)),
    "הפניקס ביטוח": ("insurance", (300, 600)),
}


def month_range(months_back: int) -> list[date]:
    """First-of-month dates for the last `months_back` full months, oldest first."""
    cursor = date.today().replace(day=1)
    months: list[date] = []
    for _ in range(months_back):
        cursor = (cursor - timedelta(days=1)).replace(day=1)
        months.append(cursor)
    return sorted(months)


def generate(months_back: int = 6, seed: int = 42) -> list[dict]:
    rng = random.Random(seed)
    rows: list[dict] = []
    for month_start in month_range(months_back):
        y, m = month_start.year, month_start.month
        days_in_month = ((month_start.replace(day=28) + timedelta(days=4)).replace(day=1) - timedelta(days=1)).day

        # salary (inflow) on the 9th
        rows.append(_row("leumi-main", date(y, m, 9), 24500.00, "משכורת חברת הייטק בעמ"))
        # rent (outflow) on the 1st
        rows.append(_row("leumi-main", date(y, m, 1), -6200.00, "שכר דירה הוראת קבע"))
        # transfer pair: main -> savings on the 10th (must net to zero)
        rows.append(_row("leumi-main", date(y, m, 10), -3000.00, "העברה לחשבון חסכון"))
        rows.append(_row("leumi-savings", date(y, m, 10), 3000.00, "העברה מחשבון עוש"))

        # card spend: 40-60 charges/month spread over both cards
        card_totals: dict[str, float] = defaultdict(float)
        for _ in range(rng.randint(40, 60)):
            merchant, (_, (lo, hi)) = rng.choice(list(MERCHANTS.items()))
            card = rng.choice(list(CARD_TO_BANK))
            amount = -round(rng.uniform(lo, hi), 2)
            rows.append(_row(card, date(y, m, rng.randint(1, days_in_month)), amount, merchant))
            card_totals[card] += amount

        # card statement debit hits the bank on the 10th of NEXT month
        debit_date = (month_start + timedelta(days=40)).replace(day=10)
        if debit_date <= date.today():
            for card, total in card_totals.items():
                rows.append(_row(CARD_TO_BANK[card], debit_date, round(total, 2), CARD_DEBIT_DESC[card]))
    return rows


def _row(account_id: str, posted: date, amount: float, desc: str) -> dict:
    return {
        "account_id": account_id,
        "account_type": ACCOUNTS[account_id],
        "posted_date": posted.isoformat(),
        "amount": f"{amount:.2f}",
        "currency": "ILS",
        "raw_description": desc,
        "source": "seed",
    }


def assign_ids(rows: list[dict]) -> None:
    """Stable id = hash(natural key + occurrence index). Prevents same-day duplicate collisions."""
    counts: dict[tuple, int] = defaultdict(int)
    for r in rows:
        key = (r["posted_date"], r["amount"], r["raw_description"], r["account_id"])
        counts[key] += 1
        raw = "|".join([*key, str(counts[key])])
        r["transaction_id"] = hashlib.sha256(raw.encode()).hexdigest()[:16]


if __name__ == "__main__":
    rows = generate()
    assign_ids(rows)
    rows.sort(key=lambda r: r["posted_date"])
    with OUT.open("w", newline="", encoding="utf-8") as f:
        writer = csv.DictWriter(f, fieldnames=list(rows[0].keys()))
        writer.writeheader()
        writer.writerows(rows)
    print(f"wrote {len(rows)} rows -> {OUT}")
```

### 1.4 `sync.py` — the one entrypoint

```python
"""kaspion sync: ingest -> dbt -> categorize. Seed mode now, scraper mode in Phase 5."""
from __future__ import annotations

import argparse
import subprocess
import sys
from pathlib import Path

from kaspion.db import connect

ROOT = Path(__file__).resolve().parent
SEED_CSV = ROOT / "dbt" / "seeds" / "sample_transactions.csv"


def load_seed() -> int:
    con = connect()
    before = con.execute("SELECT count(*) FROM raw.transactions").fetchone()[0]
    con.execute(
        f"""
        INSERT INTO raw.transactions
            (transaction_id, account_id, account_type, posted_date,
             amount, currency, raw_description, source)
        SELECT transaction_id, account_id, account_type, posted_date,
               amount, currency, raw_description, source
        FROM read_csv_auto('{SEED_CSV}', header=true)
        WHERE transaction_id NOT IN (SELECT transaction_id FROM raw.transactions)
        """
    )
    after = con.execute("SELECT count(*) FROM raw.transactions").fetchone()[0]
    con.close()
    return after - before


def run_dbt() -> None:
    subprocess.run(
        ["dbt", "build"], cwd=ROOT / "dbt",
        env={"DBT_PROFILES_DIR": ".", "PATH": __import__("os").environ["PATH"],
             "HOME": __import__("os").environ["HOME"]},
        check=True,
    )


def main() -> None:
    parser = argparse.ArgumentParser()
    parser.add_argument("--source", choices=["seed", "scraper"], default="seed")
    parser.add_argument("--skip-categorize", action="store_true")
    args = parser.parse_args()

    if args.source == "seed":
        new = load_seed()
        print(f"ingested {new} new transactions from seed")
    else:
        from kaspion.ingest.scraper_loader import load_from_scraper  # Phase 5
        new = load_from_scraper()
        print(f"ingested {new} new transactions from scraper")

    run_dbt()

    if not args.skip_categorize:
        from kaspion.ai.categorize import categorize_new_merchants  # Phase 3
        try:
            n = categorize_new_merchants()
            print(f"categorized {n} new merchants")
            if n:
                run_dbt()  # rebuild so new categories land in marts
        except ModuleNotFoundError:
            print("categorizer not built yet (Phase 3) — skipping")


if __name__ == "__main__":
    main()
```

**Done when:** `python kaspion/ingest/generate_seed.py && python sync.py --skip-categorize` runs; `SELECT count(*) FROM raw.transactions` returns 300+; re-running `sync.py` ingests 0 new rows (dedup works).

### 2.3 `dbt/models/intermediate/int_transfers_detected.sql`

```sql
-- Inter-account transfers: opposite sign, same abs amount (±1 ILS), within 3 days,
-- between the owner's own BANK accounts. Card payments are handled separately.
with bank_txns as (
    select * from {{ ref('stg_transactions') }}
    where account_type = 'bank'
),

pairs as (
    select
        o.transaction_id as out_id,
        i.transaction_id as in_id,
        row_number() over (
            partition by o.transaction_id
            order by abs(datediff('day', o.posted_date, i.posted_date)), i.transaction_id
        ) as rn_out,
        row_number() over (
            partition by i.transaction_id
            order by abs(datediff('day', o.posted_date, i.posted_date)), o.transaction_id
        ) as rn_in
    from bank_txns o
    join bank_txns i
      on o.amount < 0
     and i.amount > 0
     and abs(o.amount + i.amount) <= 1.0
     and o.account_id <> i.account_id
     and abs(datediff('day', o.posted_date, i.posted_date)) <= 3
),

matched as (
    select out_id, in_id from pairs where rn_out = 1 and rn_in = 1
)

select out_id as transaction_id, in_id as counterpart_id from matched
union all
select in_id as transaction_id, out_id as counterpart_id from matched
```

### 2.5 `dbt/models/intermediate/int_categorized.sql`

```sql
-- Resolution order: owner override ALWAYS wins > AI proposal > 'other'.
with txns as (
    select * from {{ ref('stg_transactions') }}
),

overrides as (
    select merchant_key, category_id from {{ source('state', 'merchant_overrides') }}
),

proposals as (
    select merchant_key, proposed_category_id from {{ source('state', 'ai_proposals') }}
)

select
    t.*,
    coalesce(o.category_id, p.proposed_category_id,
             case when t.amount > 0 then 'income' else 'other' end) as category_id,
    case
        when o.category_id is not null then 'override'
        when p.proposed_category_id is not null then 'ai'
        else 'default'
    end as category_source,
    tr.transaction_id is not null  as is_transfer,
    cp.transaction_id is not null  as is_card_payment
from txns t
left join overrides o  on t.merchant_key = o.merchant_key
left join proposals p  on t.merchant_key = p.merchant_key
left join {{ ref('int_transfers_detected') }} tr on t.transaction_id = tr.transaction_id
left join {{ ref('int_card_payments') }}     cp on t.transaction_id = cp.transaction_id
```

### 2.8 `dbt/models/marts/fct_budget_pacing.sql`

```sql
-- Per category x month: actual vs budget, prorated pace for the current month.
with spend as (
    select posted_month, category_id, sum(spend_ils) as actual_ils
    from {{ ref('fct_spend') }}
    group by 1, 2
),

budgets as (
    select category_id, monthly_amount_ils from {{ source('state', 'budgets') }}
),

calendar as (
    select
        posted_month,
        case
            when posted_month = date_trunc('month', current_date)
            then extract(day from current_date)::double
                 / extract(day from (date_trunc('month', current_date)
                     + interval 1 month - interval 1 day))::double
            else 1.0
        end as month_fraction
    from (select distinct posted_month from spend)
)

select
    s.posted_month,
    s.category_id,
    s.actual_ils,
    b.monthly_amount_ils                                   as budget_ils,
    round(b.monthly_amount_ils * c.month_fraction, 2)      as budget_to_date_ils,
    round(b.monthly_amount_ils * c.month_fraction - s.actual_ils, 2) as pace_ils,  -- + = under budget
    case
        when b.monthly_amount_ils is null then 'no_budget'
        when s.actual_ils <= b.monthly_amount_ils * c.month_fraction then 'under'
        else 'over'
    end as pace_status
from spend s
left join budgets b using (category_id)
join calendar c using (posted_month)
```

```sql
-- Each bank-side card debit ≈ that card's prior-month charge total (±1%).
-- Seed maps: max debit desc contains 'מקס' -> card max-1234, ישראכרט -> isracard-5678.
with debits as (
    select
        posted_date,
        date_trunc('month', posted_date - interval 1 month) as statement_month,
        case
            when raw_description like '%מקס%'     then 'max-1234'
            when raw_description like '%ישראכרט%' then 'isracard-5678'
        end as card_account,
        abs(amount) as debit_ils
    from {{ ref('int_card_payments') }}
),

card_spend as (
    select account_id as card_account, posted_month as statement_month,
           sum(abs(amount)) as charges_ils
    from {{ ref('stg_transactions') }}
    where account_type = 'credit_card'
    group by 1, 2
)

select d.card_account, d.statement_month, d.debit_ils, c.charges_ils
from debits d
join card_spend c using (card_account, statement_month)
where abs(d.debit_ils - c.charges_ils) > c.charges_ils * 0.01
```

from __future__ import annotations

import json

import requests

VALID_CATEGORIES = [
    "groceries", "restaurants", "transport", "housing", "kids", "health",
    "entertainment", "clothing", "electronics", "subscriptions", "insurance",
    "gifts", "income", "other",
]

PROMPT = """You categorize Israeli household transactions. Merchants may be Hebrew or English.
Return ONLY a JSON object mapping each merchant string to one category id from this list:
{categories}

Merchants:
{merchants}
"""


class Provider(Protocol):
    name: str
    model: str

    def categorize(self, merchants: list[str]) -> dict[str, str]: ...


class OllamaProvider:
    """Default: free, fully local. Requires `ollama pull llama3.1:8b` (or qwen2.5:7b)."""

    name = "ollama"

    def __init__(self, model: str = "llama3.1:8b", host: str = "http://127.0.0.1:11434"):
        self.model, self.host = model, host

    def categorize(self, merchants: list[str]) -> dict[str, str]:
        prompt = PROMPT.format(categories=", ".join(VALID_CATEGORIES), merchants="\n".join(merchants))
        resp = requests.post(
            f"{self.host}/api/chat",
            json={
                "model": self.model,
                "messages": [{"role": "user", "content": prompt}],
                "format": "json",
                "stream": False,
            },
            timeout=300,
        )
        resp.raise_for_status()
        return _clean(json.loads(resp.json()["message"]["content"]), merchants)


class ClaudeProvider:
    """Opt-in, paid (cents). Needs ANTHROPIC_API_KEY env var."""

    name = "claude"

    def __init__(self, model: str = "claude-haiku-4-5"):
        self.model = model

    def categorize(self, merchants: list[str]) -> dict[str, str]:
        import anthropic

        client = anthropic.Anthropic()  # reads ANTHROPIC_API_KEY
        prompt = PROMPT.format(categories=", ".join(VALID_CATEGORIES), merchants="\n".join(merchants))
        msg = client.messages.create(
            model=self.model, max_tokens=2048,
            messages=[{"role": "user", "content": prompt}],
        )
        text = msg.content[0].text
        text = text[text.index("{"): text.rindex("}") + 1]  # tolerate prose around JSON
        return _clean(json.loads(text), merchants)


class NoneProvider:
    """Manual mode: propose nothing; everything defaults to 'other' until overridden."""

    name = "none"
    model = "none"

    def categorize(self, merchants: list[str]) -> dict[str, str]:
        return {}


def _clean(raw: dict[str, str], merchants: list[str]) -> dict[str, str]:
    return {m: raw[m] for m in merchants if raw.get(m) in VALID_CATEGORIES}


def get_provider() -> Provider:
    choice = os.environ.get("KASPION_AI_PROVIDER", "ollama")
    return {"ollama": OllamaProvider, "claude": ClaudeProvider, "none": NoneProvider}[choice]()

from kaspion.ai.providers import get_provider
from kaspion.db import connect

BATCH_SIZE = 20


def categorize_new_merchants() -> int:
    """Ask the AI only about merchants with no override AND no prior proposal."""
    con = connect()
    merchants = [
        r[0] for r in con.execute(
            """
            SELECT DISTINCT s.merchant_key
            FROM main.stg_transactions s
            WHERE s.amount < 0
              AND s.merchant_key NOT IN (SELECT merchant_key FROM state.merchant_overrides)
              AND s.merchant_key NOT IN (SELECT merchant_key FROM state.ai_proposals)
            """
        ).fetchall()
    ]
    if not merchants:
        con.close()
        return 0

    provider = get_provider()
    done = 0
    for i in range(0, len(merchants), BATCH_SIZE):
        batch = merchants[i : i + BATCH_SIZE]
        for merchant, category in provider.categorize(batch).items():
            con.execute(
                """
                INSERT INTO state.ai_proposals (merchant_key, proposed_category_id, provider, model)
                VALUES (?, ?, ?, ?)
                ON CONFLICT (merchant_key) DO UPDATE
                    SET proposed_category_id = excluded.proposed_category_id
                """,
                [merchant, category, provider.name, provider.model],
            )
            done += 1
    con.close()
    return done

### 5.3 `kaspion/ingest/crypto.py` — credentials at rest

```python
from __future__ import annotations

import json
import os
from pathlib import Path

from cryptography.hazmat.primitives.ciphers.aead import AESGCM

DATA = Path(__file__).resolve().parents[2] / "data"
KEY_FILE = DATA / ".kaspion_key"
CRED_FILE = DATA / "credentials.json.enc"


def _key() -> bytes:
    if not KEY_FILE.exists():
        KEY_FILE.write_bytes(AESGCM.generate_key(bit_length=256))
        os.chmod(KEY_FILE, 0o600)
    return KEY_FILE.read_bytes()


def save_credentials(creds: dict) -> None:
    """creds example: {"leumi": {"type": "bank", "credentials": {"username": "...", "password": "..."}},
                       "max":   {"type": "credit_card", "credentials": {"username": "...", "password": "..."}}}"""
    nonce = os.urandom(12)
    blob = AESGCM(_key()).encrypt(nonce, json.dumps(creds).encode(), None)
    CRED_FILE.write_bytes(nonce + blob)
    os.chmod(CRED_FILE, 0o600)


def load_credentials() -> dict:
    raw = CRED_FILE.read_bytes()
    return json.loads(AESGCM(_key()).decrypt(raw[:12], raw[12:], None))
```

### 5.4 `kaspion/ingest/scraper_loader.py`

```python
from __future__ import annotations

import hashlib
import json
import subprocess
from collections import defaultdict
from datetime import date, timedelta
from pathlib import Path

from kaspion.db import connect
from kaspion.ingest.crypto import load_credentials

SCRAPER_DIR = Path(__file__).resolve().parents[2] / "scraper"


def load_from_scraper(days_back: int = 60) -> int:
    creds = load_credentials()
    all_rows: list[dict] = []
    for company, cfg in creds.items():
        proc = subprocess.run(
            ["node", "scrape.js"],
            cwd=SCRAPER_DIR,
            env={
                "KASPION_COMPANY": company,
                "KASPION_CREDENTIALS": json.dumps(cfg["credentials"]),
                "KASPION_ACCOUNT_TYPE": cfg["type"],
                "KASPION_START_DATE": (date.today() - timedelta(days=days_back)).isoformat(),
                "PATH": __import__("os").environ["PATH"],
                "HOME": __import__("os").environ["HOME"],
            },
            capture_output=True, text=True, check=True,
        )
        all_rows.extend(json.loads(proc.stdout))

    _assign_ids(all_rows)
    con = connect()
    before = con.execute("SELECT count(*) FROM raw.transactions").fetchone()[0]
    con.executemany(
        """
        INSERT INTO raw.transactions
            (transaction_id, account_id, account_type, posted_date,
             amount, currency, raw_description, source)
        VALUES (?, ?, ?, ?, ?, ?, ?, ?)
        ON CONFLICT (transaction_id) DO NOTHING
        """,
        [
            [r["transaction_id"], r["account_id"], r["account_type"], r["posted_date"],
             r["amount"], r["currency"], r["raw_description"], r["source"]]
            for r in all_rows
        ],
    )
    after = con.execute("SELECT count(*) FROM raw.transactions").fetchone()[0]
    con.close()
    return after - before


def _assign_ids(rows: list[dict]) -> None:
    counts: dict[tuple, int] = defaultdict(int)
    for r in rows:
        if r.get("source_id"):  # prefer the bank's own id
            r["transaction_id"] = hashlib.sha256(
                f"{r['account_id']}|{r['source_id']}".encode()
            ).hexdigest()[:16]
        else:
            key = (r["posted_date"], str(r["amount"]), r["raw_description"], r["account_id"])
            counts[key] += 1
            r["transaction_id"] = hashlib.sha256(
                "|".join([*key, str(counts[key])]).encode()
            ).hexdigest()[:16]
```

**`evals/run_evals.py`**

```python
"""Score AI proposals vs hand-labeled ground truth. Usage:
   python evals/run_evals.py            # score current state.ai_proposals
   python evals/run_evals.py --provider ollama --provider claude   # compare (re-runs categorization per provider)
"""
from __future__ import annotations

import argparse
import json
import os
from pathlib import Path

import pandas as pd

from kaspion.db import connect

HERE = Path(__file__).resolve().parent


def score_current() -> dict:
    con = connect(read_only=True)
    proposals = con.execute(
        "SELECT merchant_key, proposed_category_id, provider, model FROM state.ai_proposals"
    ).df()
    con.close()
    truth = pd.read_csv(HERE / "ground_truth.csv")
    merged = truth.merge(proposals, on="merchant_key", how="left")
    merged["correct"] = merged["true_category_id"] == merged["proposed_category_id"]

    per_cat = (
        merged.groupby("true_category_id")
        .agg(n=("correct", "size"), recall=("correct", "mean"))
        .reset_index()
    )
    precision = (
        merged.dropna(subset=["proposed_category_id"])
        .groupby("proposed_category_id")["correct"].mean()
        .rename("precision").reset_index()
    )
    report = {
        "provider": proposals["provider"].mode().iat[0] if len(proposals) else "none",
        "model": proposals["model"].mode().iat[0] if len(proposals) else "none",
        "n_labeled": int(len(merged)),
        "coverage": float(merged["proposed_category_id"].notna().mean()),
        "accuracy": float(merged["correct"].mean()),
        "per_category": per_cat.merge(
            precision, left_on="true_category_id", right_on="proposed_category_id", how="left"
        )[["true_category_id", "n", "recall", "precision"]].to_dict("records"),
        "confusion": merged[~merged["correct"]][
            ["merchant_key", "true_category_id", "proposed_category_id"]
        ].to_dict("records"),
    }
    return report


def main() -> None:
    parser = argparse.ArgumentParser()
    parser.add_argument("--provider", action="append", default=None)
    args = parser.parse_args()

    reports = []
    if args.provider:
        from kaspion.ai.categorize import categorize_new_merchants
        for p in args.provider:
            con = connect(); con.execute("DELETE FROM state.ai_proposals"); con.close()
            os.environ["KASPION_AI_PROVIDER"] = p
            categorize_new_merchants()
            reports.append(score_current())
    else:
        reports.append(score_current())

    for r in reports:
        print(f"\n=== {r['provider']} / {r['model']} ===")
        print(f"accuracy: {r['accuracy']:.1%}   coverage: {r['coverage']:.1%}   n={r['n_labeled']}")
        print(pd.DataFrame(r["per_category"]).to_string(index=False))
        if r["confusion"]:
            print("misses:")
            print(pd.DataFrame(r["confusion"]).to_string(index=False))
    (HERE / "last_run.json").write_text(json.dumps(reports, ensure_ascii=False, indent=2))


if __name__ == "__main__":
    main()
```

**`kaspion/semantic/resolver.py`**

```python
from __future__ import annotations

from pathlib import Path

import yaml

from kaspion.db import connect

SPEC = yaml.safe_load((Path(__file__).parent / "metrics.yml").read_text())


def query_metric(
    metric: str,
    group_by: list[str] | None = None,
    filters: dict[str, str] | None = None,
) -> list[dict]:
    """Compile a metric+dimensions request to SQL and run it read-only.
    Only known metrics/dimensions are accepted -> no SQL injection surface."""
    m = SPEC["metrics"][metric]
    group_by = group_by or []
    dims = []
    for d in group_by:
        dim = SPEC["dimensions"][d]
        if m["table"] not in dim["tables"]:
            raise ValueError(f"dimension {d} not available on {m['table']}")
        dims.append(f"{dim['sql']} as {d}")

    where = ["1=1"]
    params: list[str] = []
    for d, val in (filters or {}).items():
        dim = SPEC["dimensions"][d]
        where.append(f"cast({dim['sql']} as varchar) = ?")
        params.append(val)

    select = ", ".join([*dims, f"{m['sql']} as {metric}"])
    group = f"group by {', '.join(str(i + 1) for i in range(len(dims)))}" if dims else ""
    sql = f"select {select} from {m['table']} where {' and '.join(where)} {group}"

    con = connect(read_only=True)
    out = con.execute(sql, params).df().to_dict("records")
    con.close()
    return out
```

## Phase 8 — LangGraph monthly-review agent

**`kaspion/agent/review.py`**

```python
"""Monthly review: pull month via semantic layer -> detect anomalies -> draft Hebrew summary.
Usage: python -m kaspion.agent.review --month 2026-06-01"""
from __future__ import annotations

import argparse
import statistics
from typing import TypedDict

from langgraph.graph import END, StateGraph

from kaspion.semantic.resolver import query_metric


class ReviewState(TypedDict):
    month: str
    spend: list[dict]
    pace: list[dict]
    anomalies: list[str]
    summary: str


def fetch(state: ReviewState) -> dict:
    return {
        "spend": query_metric("total_spend", ["category", "month"]),
        "pace": query_metric("budget_pace", ["category", "month"], {"month": state["month"]}),
    }


def analyze(state: ReviewState) -> dict:
    """Anomaly = this month's category spend > median + 2*MAD of prior months. Overspend = pace < 0."""
    anomalies = []
    by_cat: dict[str, list[tuple[str, float]]] = {}
    for r in state["spend"]:
        by_cat.setdefault(r["category"], []).append((str(r["month"]), r["total_spend"]))
    for cat, series in by_cat.items():
        history = [v for mth, v in series if mth < state["month"]]
        current = next((v for mth, v in series if mth.startswith(state["month"][:7])), None)
        if current is None or len(history) < 3:
            continue
        med = statistics.median(history)
        mad = statistics.median(abs(v - med) for v in history) or 1.0
        if current > med + 2 * mad:
            anomalies.append(f"{cat}: ₪{current:,.0f} החודש מול חציון ₪{med:,.0f}")
    for r in state["pace"]:
        if r["budget_pace"] < 0:
            anomalies.append(f"{r['category']}: חריגה של ₪{-r['budget_pace']:,.0f} מהתקציב")
    return {"anomalies": anomalies}


def summarize(state: ReviewState) -> dict:
    total = sum(
        r["total_spend"] for r in state["spend"] if str(r["month"]).startswith(state["month"][:7])
    )
    lines = [f"סיכום חודשי — {state['month'][:7]}", f"סך הוצאות: ₪{total:,.0f}"]
    lines += ["", "⚠️ דורש תשומת לב:"] + [f"• {a}" for a in state["anomalies"]] if state["anomalies"] \
        else ["", "✅ אין חריגות מהתקציב החודש."]
    return {"summary": "\n".join(lines)}


graph = StateGraph(ReviewState)
graph.add_node("fetch", fetch)
graph.add_node("analyze", analyze)
graph.add_node("summarize", summarize)
graph.set_entry_point("fetch")
graph.add_edge("fetch", "analyze")
graph.add_edge("analyze", "summarize")
graph.add_edge("summarize", END)
app = graph.compile()

if __name__ == "__main__":
    p = argparse.ArgumentParser()
    p.add_argument("--month", required=True, help="YYYY-MM-01")
    print(app.invoke({"month": p.parse_args().month})["summary"])
```
